# Расписание маркетингового контура corebridge.ru
# Ставится под пользователем deploy:  crontab marketing/scripts/crontab.txt
# Проверить, что встало:              crontab -l
#
# ⚠️ Часы сервера — UTC (проверено `timedatectl`, зона Etc/UTC). Москва = UTC+3,
# поэтому слева стоит UTC, а в комментарии — привычное московское время.
# Если когда-нибудь переведёшь сервер на Europe/Moscow, эти строки надо сдвинуть
# на 3 часа вперёд, иначе всё поедет.
#
# Логи nginx ротируются в 06:25 UTC (стандартный /etc/cron.daily), поэтому сбор
# стоит раньше — иначе сутки данных теряются безвозвратно.

SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
MKT=/opt/corebridge-site/marketing/scripts

# ── Сбор и сторож: работают всегда, языковая модель не участвует ────────────

# Каждые 15 минут: сайт жив, 5xx, битые ссылки, новая заявка/регистрация/оплата
*/15 * * * * $MKT/run.sh watch

# 02:40 UTC = 05:40 МСК — выгрузка за вчера
40 2 * * * $MKT/run.sh collect

# 03:00 UTC = 06:00 МСК — сводка цифрами в Telegram
0 3 * * * $MKT/run.sh digest

# ── Прогоны агента: пропускаются молча, если claude CLI не установлен ────────

# 03:05 UTC = 06:05 МСК ежедневно — разбор аномалий и замеры активных гипотез
5 3 * * * $MKT/run.sh agent 10_daily_metrics

# 04:00 UTC = 07:00 МСК. Пн/ср/пт — контент и работа по гипотезам,
# вт/чт — техническая работа: индексация, скорость, разметка.
0 4 * * 1,3,5 $MKT/run.sh agent 20_daily_content
0 4 * * 2,4   $MKT/run.sh agent 25_daily_technical

# 05:00 UTC = 08:00 МСК понедельник — итоги недели, вердикты, новые гипотезы
0 5 * * 1 $MKT/run.sh agent 30_weekly_review

# 06:00 UTC = 09:00 МСК первого числа — ревизия базы знаний и стратегии
0 6 1 * * $MKT/run.sh agent 40_monthly_revision
